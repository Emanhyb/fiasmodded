--[[
	Dummy Follow  ·  Delta Executor
	How to run: open Delta → paste this whole script → Execute.

	Enable     — while ON, if Test_Dummy is more than 3-5 studs from
	             you, walk toward it until it is 1 stud from your torso.
	Automatic  — while ON (and Enable is ON), every 2 minutes check
	             whether the dummy has drifted ~3-5 studs away.
	             If it has, walk in until you are 1 stud from your torso.
	X          — hides the panel. Tap the chip to reopen it.
	Aim Lock   — standalone toggle. While ON, lock the camera onto
	             Test_Dummy.Head. No distance or Enable requirement.
	Auto Click — standalone toggle. While ON, repeatedly left-clicks
	             the center of the screen and keeps the cursor locked
	             there. Speed slider sets clicks per second (1–20 CPS).
	             If the dummy is more than 3-5 studs out: unequip
	             Backpack.Fight (Tool 1) and pause clicking. When
	             the dummy is in range: re-equip Tool 1 and resume clicking.

	Respawn   — First spawn is workspace.Live.Test_Dummy. After it
	             respawns (private server) the path is workspace.Test_Dummy.
	             Both are watched. If Enable is ON, walk back to 1 stud.
]]

local function isTestDummyName(name)
	if not name then
		return false
	end
	local n = string.lower(name)
	return n == "test_dummy" or n == "testdummy"
end

local function isDummyAlive(model)
	if not model or not model.Parent then
		return false
	end
	if not model:IsDescendantOf(workspaceRef) then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health <= 0 then
		return false
	end
	return true
end

local function considerDummy(inst, list, seen)
	if not inst or seen[inst] then
		return
	end
	if not isTestDummyName(inst.Name) then
		return
	end
	seen[inst] = true
	list[#list + 1] = inst
end

-- First spawn: workspace.Live.Test_Dummy
-- After first private-server respawn: workspace.Test_Dummy
local function getDummyModel()
	local list = {}
	local seen = {}
	considerDummy(workspaceRef:FindFirstChild("Test_Dummy"), list, seen)
	local live = getLive()
	if live then
		considerDummy(live:FindFirstChild("Test_Dummy"), list, seen)
		for _, child in ipairs(live:GetChildren()) do
			considerDummy(child, list, seen)
		end
	end
	for _, child in ipairs(workspaceRef:GetChildren()) do
		considerDummy(child, list, seen)
	end
	for i = 1, #list do
		if isDummyAlive(list[i]) then
			return list[i]
		end
	end
	if #list > 0 then
		return list[1]
	end
	return nil
end

local function planarDistance(a, b)
	local dx = a.X - b.X
	local dz = a.Z - b.Z
	return math.sqrt(dx * dx + dz * dz)
end

local function currentDistance()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()
	if not torso or not dummy then
		return nil
	end
	return planarDistance(torso.Position, dummy.Position)
end

	return player:FindFirstChild("Backpack")

local function isFightTool(inst)
	if not inst or not inst:IsA("Tool") then
		return false
	end
	local n = string.lower(tostring(inst.Name))
	return n == "fight" or n == "1" or n == "tool 1" or n == "tool1"
end

-- Fight is hotbar Tool 1. Prefer that name, else the first Tool in Backpack.
local function getFightTool()
	local char = getCharacter()
	if char then
		for _, child in ipairs(char:GetChildren()) do
			if isFightTool(child) then
				return child
			end
		end
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") then
				return child
			end
		end
	end
	local bag = getBackpack()
	if bag then
		local named = bag:FindFirstChild("Fight")
		if named and named:IsA("Tool") then
			return named
		end
		for _, child in ipairs(bag:GetChildren()) do
			if child:IsA("Tool") then
				return child
			end
		end
	end
	return nil
end

local function equippedTool()
	local char = getCharacter()
	if not char then
		return nil
	end
	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Tool") then
			return child
		end
	end
	return nil
end

local function fightIsEquipped()
	local char = getCharacter()
	if not char then
		return false
	end
	for _, child in ipairs(char:GetChildren()) do
		if isFightTool(child) then
			return true
		end
	end
	return false
end

-- ONLY called when dummy is more than 3-5 studs out. Never while in range.
local function unequipFight()
	local hum = getHumanoid()
	if hum then
		pcall(function()
			hum:UnequipTools()
		end)
	end
	local char = getCharacter()
	local bag = getBackpack()
	if char and bag then
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") then
				pcall(function()
					child.Parent = bag
				end)
			end
		end
	end
end

-- Press hotbar 1 exactly once. Never spam — 1 toggles Tool 1 off if held.
local function pressKey1Once()
	pcall(function()
		if VirtualInputManager then
			VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.One, false, game)
			VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.One, false, game)
		end
	end)
	if typeof(keypress) == "function" then
		pcall(keypress, 0x31)
		if typeof(keyrelease) == "function" then
			pcall(keyrelease, 0x31)
		end
	end
end

-- true = in range, false = clearly out, nil = unknown (do not unequip)
local function dummyRangeState()
	local dist = currentDistance()
	if dist == nil then
		return nil
	end
	if dist <= FOLLOW_MAX then
		return true
	end
	return false
end

local function updateFightAndClick()
	if not running then
		clickAllowed = false
		return
	end
	if not (enabled or clickerOn) then
		clickAllowed = false
		return
	end

	local inRange = dummyRangeState()
	if inRange == true then
		-- Entering range: press 1 once to equip Tool 1. Do not press again.
		if fightState ~= "in" then
			if not fightIsEquipped() then
				pressKey1Once()
			end
			fightState = "in"
		end
		clickAllowed = clickerOn
	elseif inRange == false then
		if fightState ~= "out" or fightIsEquipped() or equippedTool() then
			unequipFight()
		end
		fightState = "out"
		clickAllowed = false
	else
		if fightState == "out" then
			clickAllowed = false
		else
			clickAllowed = clickerOn
		end
	end
end

local function pickFollowTrigger()
	return FOLLOW_MIN + math.random() * (FOLLOW_MAX - FOLLOW_MIN)
end

local function pickAutoDrift()
	return AUTO_DRIFT_MIN + math.random() * (AUTO_DRIFT_MAX - AUTO_DRIFT_MIN)
end

----------------------------------------------------------------------
-- Movement  (Humanoid:Move every frame — MoveTo is often cancelled)
----------------------------------------------------------------------
local function haltHumanoid()
	local humanoid = getHumanoid()
	if humanoid then
		pcall(function()
			humanoid:Move(Vector3.zero)
		end)
	end
end

local function stopWalking()
	walking = false
	haltHumanoid()
end

local function walkUntilClose(reason)
	walking = true
	lastWalkReason = reason
	lastMoveToAt = 0
end

local function stepWalk()
	if not walking or not enabled then
		return
	end

	local humanoid = getHumanoid()
	local root = getRoot()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()

	if not humanoid or humanoid.Health <= 0 or not root or not torso then
		walking = false
		haltHumanoid()
		return
	end

	if not dummy then
		waitingRespawn = true
		lastWalkReason = "waiting respawn"
		walking = false
		haltHumanoid()
		return
	end

	local dist = planarDistance(torso.Position, dummy.Position)
	if dist <= STOP_DISTANCE then
		haltHumanoid()
		walking = false
		lastWalkReason = "arrived"
		return
	end

	local flat = Vector3.new(
		dummy.Position.X - root.Position.X,
		0,
		dummy.Position.Z - root.Position.Z
	)
	local mag = flat.Magnitude
	if mag > 0.05 then
		pcall(function()
			humanoid:Move(flat.Unit)
		end)
		local now = os.clock()
		if now - lastMoveToAt > 0.25 then
			lastMoveToAt = now
			pcall(function()
				humanoid:MoveTo(dummy.Position)
			end)
		end
	end
end

local function stepAimLock()
	if not running or not aimLockOn then
		aimEngaged = false
		return
	end

	local cam = workspaceRef.CurrentCamera
	local head = getDummyHead()
	if not cam or not head then
		aimEngaged = false
		return
	end

	aimEngaged = true
	local lookAt = head.Position
	pcall(function()
		cam.CFrame = CFrame.lookAt(cam.CFrame.Position, lookAt)
	end)

	local root = getRoot()
	if root then
		local flat = Vector3.new(lookAt.X, root.Position.Y, lookAt.Z)
		if (flat - root.Position).Magnitude > 0.08 then
			pcall(function()
				root.CFrame = CFrame.lookAt(root.Position, flat)
			end)
		end
	end
end

----------------------------------------------------------------------
-- Dummy respawn watcher
----------------------------------------------------------------------
local function onDummyLost(reason)
	seenDummy = nil
	waitingRespawn = true
	lastDummyPos = nil
	if walking then
		walking = false
		haltHumanoid()
	end
	aimEngaged = false
	lastWalkReason = reason or "waiting respawn"
end

local function approachAfterRespawn(model)
	approachToken = approachToken + 1
	local myToken = approachToken
	task.spawn(function()
		local deadline = os.clock() + RESPAWN_WAIT
		local part = nil
		while running and myToken == approachToken and os.clock() < deadline do
			if not model or not model.Parent then
				return
			end
			part = model:FindFirstChild("LowerTorso")
			if part then
				break
			end
			task.wait(0.1)
		end
		if myToken ~= approachToken or not running then
			return
		end
		if not part then
			part = dummyAimPart(model)
		end
		if not part then
			return
		end
		task.wait(RESPAWN_SETTLE)
		if myToken ~= approachToken or not running or not enabled then
			return
		end
		if not model.Parent or not isDummyAlive(model) then
			return
		end
		waitingRespawn = false
		seenDummy = model
		lastDummyPos = part.Position
		walkUntilClose("respawn")
	end)
end

local function hookDummy(model)
	disconnectList(dummyHookConns)
	dummyHookConns = {}
	if not model then
		return
	end
	seenDummy = model
	waitingRespawn = false

	local function add(conn)
		if conn then
			table.insert(dummyHookConns, conn)
			track(conn)
		end
	end

	add(model.AncestryChanged:Connect(function(_, parent)
		if not parent then
			onDummyLost("dummy removed")
			return
		end
		if not model:IsDescendantOf(workspaceRef) then
			onDummyLost("dummy removed")
		end
	end))

	pcall(function()
		add(model.Destroying:Connect(function()
			onDummyLost("dummy destroyed")
		end))
	end)

	local function hookHumanoid(hum)
		if not hum then
			return
		end
		add(hum.Died:Connect(function()
			onDummyLost("dummy died")
		end))
		add(hum.HealthChanged:Connect(function(h)
			if h <= 0 then
				onDummyLost("dummy died")
			end
		end))
	end

	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum then
		hookHumanoid(hum)
	else
		add(model.ChildAdded:Connect(function(ch)
			if ch:IsA("Humanoid") then
				hookHumanoid(ch)
			end
		end))
	end
end

local function hookLive(live)
	disconnectList(liveHookConns)
	liveHookConns = {}
	if not live then
		return
	end

	local function add(conn)
		if conn then
			table.insert(liveHookConns, conn)
			track(conn)
		end
	end

	add(live.ChildAdded:Connect(function(child)
		if not isTestDummyName(child.Name) then
			return
		end
		task.wait()
		hookDummy(child)
		if enabled then
			approachAfterRespawn(child)
		end
	end))

	add(live.ChildRemoved:Connect(function(child)
		if isTestDummyName(child.Name) then
			task.wait()
			if not getDummyModel() then
				onDummyLost("dummy removed")
			end
		end
	end))

	local existing = getDummyModel()
	if existing and existing.Parent == live then
		hookDummy(existing)
	end
end

local workspaceDummyConns = {}

local function hookWorkspaceDummySpawns()
	disconnectList(workspaceDummyConns)
	workspaceDummyConns = {}
	local function add(conn)
		if conn then
			table.insert(workspaceDummyConns, conn)
			track(conn)
		end
	end
	add(workspaceRef.ChildAdded:Connect(function(ch)
		if ch.Name == "Live" then
			hookLive(ch)
			return
		end
		if isTestDummyName(ch.Name) then
			task.wait()
			hookDummy(ch)
			if enabled then
				approachAfterRespawn(ch)
			end
		end
	end))
	add(workspaceRef.ChildRemoved:Connect(function(ch)
		if isTestDummyName(ch.Name) then
			task.wait()
			if not getDummyModel() then
				onDummyLost("dummy removed")
			end
		end
	end))
end

do
	local liveNow = getLive()
	if liveNow then
		hookLive(liveNow)
	end
	hookWorkspaceDummySpawns()
	local rootDummy = workspaceRef:FindFirstChild("Test_Dummy")
	if rootDummy then
		hookDummy(rootDummy)
	end
end

local function guiParent()
	if typeof(gethui) == "function" then
		local ok, hui = pcall(gethui)
		if ok and hui then
			return hui
		end
	end

	local cg = svc("CoreGui")
	local canCore = false
	pcall(function()
		local probe = Instance.new("Folder")
		probe.Name = "DF_Probe"
		probe.Parent = cg
		probe:Destroy()
		canCore = true
	end)
	if canCore then
		return cg
	end

	local pg = player:FindFirstChild("PlayerGui")
	if not pg then
		pg = player:WaitForChild("PlayerGui", 5)
	end
	return pg
end

local function destroyOld(parent)
	if not parent then
		return
	end
	for _, child in ipairs(parent:GetChildren()) do
		if child.Name == "DummyFollowGui" then
			pcall(function()
				child:Destroy()
			end)
		end
	end
end

pcall(function()
	if typeof(gethui) == "function" then
		destroyOld(gethui())
	end
end)
pcall(function()
	destroyOld(svc("CoreGui"))
end)
pcall(function()
	local pg = player:FindFirstChild("PlayerGui")
	if pg then
		destroyOld(pg)
	end
end)

local function clickInterval()
	return 1 / math.max(clickCps, 1)
end

local function setClickSpeedFromAlpha(alpha)
	if alpha < 0 then
		alpha = 0
	elseif alpha > 1 then
		alpha = 1
	end
	clickCps = math.floor(CLICK_CPS_MIN + alpha * (CLICK_CPS_MAX - CLICK_CPS_MIN) + 0.5)
	if clickCps < CLICK_CPS_MIN then
		clickCps = CLICK_CPS_MIN
	elseif clickCps > CLICK_CPS_MAX then
		clickCps = CLICK_CPS_MAX
	end
	speedValue.Text = tostring(clickCps) .. " CPS"
	speedFill.Size = UDim2.new(alpha, 0, 1, 0)
	speedKnob.Position = UDim2.new(alpha, 0, 0.5, 0)
	if clickerOn then
		clickHint.Text = "Clicking  ·  " .. tostring(clickCps) .. " CPS"
	end
end

setClickSpeedFromAlpha((clickCps - CLICK_CPS_MIN) / (CLICK_CPS_MAX - CLICK_CPS_MIN))

do
	local sliding = false

	local function setFromX(x)
		local absX = speedTrack.AbsolutePosition.X
		local w = speedTrack.AbsoluteSize.X
		if w < 1 then
			return
		end
		setClickSpeedFromAlpha((x - absX) / w)
	end

	local function beginSlide(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			if panelDrag then
				pcall(function()
					panelDrag.Enabled = false
				end)
			end
			setFromX(input.Position.X)
		end
	end

	local function moveSlide(input)
		if not sliding then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			setFromX(input.Position.X)
		end
	end

	local function endSlide(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
			if panelDrag then
				pcall(function()
					panelDrag.Enabled = true
				end)
			end
		end
	end

	track(speedTrack.InputBegan:Connect(beginSlide))
	track(speedKnob.InputBegan:Connect(beginSlide))
	track(UserInputService.InputChanged:Connect(moveSlide))
	track(UserInputService.InputEnded:Connect(endSlide))
	pcall(function()
		track(UserInputService.TouchMoved:Connect(moveSlide))
		track(UserInputService.TouchEnded:Connect(endSlide))
	end)
end

	local function beginDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = panel.Position
		end
	end

	local function updateDrag(input)
		if not dragging then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			local d = input.Position - dragStart
			panel.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end

	local function endDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end

	track(header.InputBegan:Connect(beginDrag))
	track(UserInputService.InputChanged:Connect(updateDrag))
	track(UserInputService.InputEnded:Connect(endDrag))
	pcall(function()
		track(UserInputService.TouchMoved:Connect(updateDrag))
		track(UserInputService.TouchEnded:Connect(endDrag))
	end)

local function pointOver(guiObj, point)
	if not guiObj or not guiObj.Visible then
		return false
	end
	local p = guiObj.AbsolutePosition
	local s = guiObj.AbsoluteSize
	return point.X >= p.X and point.X <= p.X + s.X
		and point.Y >= p.Y and point.Y <= p.Y + s.Y
end

local function screenCenter()
	local cam = workspaceRef.CurrentCamera
	if cam then
		return cam.ViewportSize * 0.5
	end
	return Vector2.new(400, 300)
end

local function lockCursorCenter()
	local pos = screenCenter()
	if typeof(mousemoveabs) == "function" then
		pcall(mousemoveabs, pos.X, pos.Y)
	end
	return pos
end

local function centerOverOurGui()
	local loc = screenCenter()
	local inset = Vector2.new(0, 0)
	pcall(function()
		inset = GuiService:GetGuiInset()
	end)
	local adj = Vector2.new(loc.X - inset.X, loc.Y - inset.Y)
	if uiOpen and panel.Visible and (pointOver(panel, loc) or pointOver(panel, adj)) then
		return true
	end
	if restore.Visible and (pointOver(restore, loc) or pointOver(restore, adj)) then
		return true
	end
	return false
end

local function fireClick()
	if not clickerOn or not running or not clickAllowed then
		return
	end
	local pos = lockCursorCenter()
	if centerOverOurGui() then
		return
	end

	if VirtualInputManager then
		pcall(function()
			VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 1)
			VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 1)
		end)
		return
	end
	if typeof(mouse1click) == "function" then
		pcall(mouse1click)
		return
	end
	if typeof(mouse1press) == "function" then
		pcall(mouse1press)
		task.wait(0.015)
		if typeof(mouse1release) == "function" then
			pcall(mouse1release)
		end
	end
end

task.spawn(function()
	local trigger = pickFollowTrigger()
	while running and gui.Parent do
		if enabled and not walking then
			local ok, dist = pcall(currentDistance)
			if ok and dist and dist > trigger then
				walkUntilClose(string.format("enable  %.0f studs", trigger))
				trigger = pickFollowTrigger()
			elseif ok and dist and dist <= STOP_DISTANCE then
				trigger = pickFollowTrigger()
			end
		end
		task.wait(SCAN_RATE)
	end
end)

task.spawn(function()
	while running and gui.Parent do
		task.wait(0.25)
		if automatic and enabled and os.clock() >= nextAutoAt then
			nextAutoAt = os.clock() + AUTO_INTERVAL

			local dummy = getDummyLowerTorso()
			local torso = getPlayerTorso()
			if dummy and torso then
				local dist = planarDistance(torso.Position, dummy.Position)
				local moved = 0
				if lastDummyPos then
					moved = (dummy.Position - lastDummyPos).Magnitude
				end
				lastDummyPos = dummy.Position

				local driftNeed = pickAutoDrift()
				if dist >= driftNeed or moved >= driftNeed then
					walkUntilClose(string.format("auto  %.0f studs", driftNeed))
				end
			end
		end
	end
end)

-- Backup poll in case ChildAdded is missed after a respawn
task.spawn(function()
	while running and gui.Parent do
		local model = getDummyModel()
		local alive = model and isDummyAlive(model)
		if not alive then
			if not waitingRespawn then
				onDummyLost("dummy missing")
			end
		else
			if model ~= seenDummy then
				hookDummy(model)
				if enabled then
					approachAfterRespawn(model)
				end
			elseif waitingRespawn then
				if enabled then
					approachAfterRespawn(model)
				else
					waitingRespawn = false
				end
			end
		end
		task.wait(0.35)
	end
end)

task.spawn(function()
	while running and gui.Parent do
		if clickerOn then
			pcall(fireClick)
			task.wait(clickInterval())
		else
			task.wait(0.12)
		end
	end
end)
