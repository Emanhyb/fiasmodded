--[[
	Dummy Follow  ·  Delta Executor
	How to run: open Delta → paste this whole script → Execute.

	Enable     — while ON, if workspace.Live.Test_Dummy.LowerTorso is
	             more than 10-20 studs from you, walk toward it until
	             it is 1 stud from your torso.
	Automatic  — while ON (and Enable is ON), every 2 minutes check
	             whether the dummy has drifted ~10-15 studs away.
	             If it has, walk in until you are 1 stud from your torso.
	X          — hides the panel. Tap the chip to reopen it.

	Respawn   — Test_Dummy is watched on workspace.Live. When it dies
	             or is replaced, the script waits and, if Enable is ON,
	             walks back to the new LowerTorso until you are 1 stud away.
]]

local function getPlayerTorso()
	local char = getCharacter()
	if not char then
		return nil
	end
	return char:FindFirstChild("UpperTorso")
		or char:FindFirstChild("Torso")
		or char:FindFirstChild("LowerTorso")

	return workspaceRef:FindFirstChild("Live")

local function isTestDummyName(name)
	if not name then
		return false
	end
	local n = string.lower(name)
	return n == "test_dummy" or n == "testdummy"
end

local function getDummyModel()
	local live = getLive()
	if not live then
		return nil
	end
	local dummy = live:FindFirstChild("Test_Dummy")
	if dummy then
		return dummy
	end
	for _, child in ipairs(live:GetChildren()) do
		if isTestDummyName(child.Name) then
			return child
		end
	end
	return nil
end

local function isDummyAlive(model)
	if not model or not model.Parent then
		return false
	end
	local live = getLive()
	if live and (not model:IsDescendantOf(live)) then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health <= 0 then
		return false
	end
	return true
end

-- Prefer LowerTorso. Fall back only while the new rig is still streaming in.
local function dummyAimPart(model)
	if not model then
		return nil
	end
	return model:FindFirstChild("LowerTorso")
		or model:FindFirstChild("HumanoidRootPart")
		or model:FindFirstChild("Torso")
		or model:FindFirstChild("UpperTorso")
end

local function getDummyLowerTorso()
	local dummy = getDummyModel()
	if not dummy or not isDummyAlive(dummy) then
		return nil
	end
	return dummy:FindFirstChild("LowerTorso") or dummyAimPart(dummy)
end

local function planarDistance(a, b)
	local dx = a.X - b.X
	local dz = a.Z - b.Z
	return math.sqrt(dx * dx + dz * dz)
end

local function currentDistance()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()
	if not torso or not dummy then
		return nil
	end
	return planarDistance(torso.Position, dummy.Position)
end

local function pickFollowTrigger()
	return FOLLOW_MIN + math.random() * (FOLLOW_MAX - FOLLOW_MIN)
end

local function pickAutoDrift()
	return AUTO_DRIFT_MIN + math.random() * (AUTO_DRIFT_MAX - AUTO_DRIFT_MIN)
end

----------------------------------------------------------------------
-- Movement  (Humanoid:Move every frame — MoveTo is often cancelled)
----------------------------------------------------------------------
local function haltHumanoid()
	local humanoid = getHumanoid()
	if humanoid then
		pcall(function()
			humanoid:Move(Vector3.zero)
		end)
	end
end

local function stopWalking()
	walking = false
	haltHumanoid()
end

local function walkUntilClose(reason)
	walking = true
	lastWalkReason = reason
	lastMoveToAt = 0
end

local function stepWalk()
	if not walking or not enabled then
		return
	end

	local humanoid = getHumanoid()
	local root = getRoot()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()

	if not humanoid or humanoid.Health <= 0 or not root or not torso then
		walking = false
		haltHumanoid()
		return
	end

	if not dummy then
		waitingRespawn = true
		lastWalkReason = "waiting respawn"
		walking = false
		haltHumanoid()
		return
	end

	local dist = planarDistance(torso.Position, dummy.Position)
	if dist <= STOP_DISTANCE then
		haltHumanoid()
		walking = false
		lastWalkReason = "arrived"
		return
	end

	local flat = Vector3.new(
		dummy.Position.X - root.Position.X,
		0,
		dummy.Position.Z - root.Position.Z
	)
	local mag = flat.Magnitude
	if mag > 0.05 then
		pcall(function()
			humanoid:Move(flat.Unit)
		end)
		local now = os.clock()
		if now - lastMoveToAt > 0.25 then
			lastMoveToAt = now
			pcall(function()
				humanoid:MoveTo(dummy.Position)
			end)
		end
	end
end

----------------------------------------------------------------------
-- Dummy respawn watcher
----------------------------------------------------------------------
local function onDummyLost(reason)
	seenDummy = nil
	waitingRespawn = true
	lastDummyPos = nil
	if walking then
		walking = false
		haltHumanoid()
	end
	lastWalkReason = reason or "waiting respawn"
end

local function approachAfterRespawn(model)
	approachToken = approachToken + 1
	local myToken = approachToken
	task.spawn(function()
		local deadline = os.clock() + RESPAWN_WAIT
		local part = nil
		while running and myToken == approachToken and os.clock() < deadline do
			if not model or not model.Parent then
				return
			end
			part = model:FindFirstChild("LowerTorso")
			if part then
				break
			end
			task.wait(0.1)
		end
		if myToken ~= approachToken or not running then
			return
		end
		if not part then
			part = dummyAimPart(model)
		end
		if not part then
			return
		end
		task.wait(RESPAWN_SETTLE)
		if myToken ~= approachToken or not running or not enabled then
			return
		end
		if not model.Parent or not isDummyAlive(model) then
			return
		end
		waitingRespawn = false
		seenDummy = model
		lastDummyPos = part.Position
		walkUntilClose("respawn")
	end)
end

local function hookDummy(model)
	disconnectList(dummyHookConns)
	dummyHookConns = {}
	if not model then
		return
	end
	seenDummy = model
	waitingRespawn = false

	local function add(conn)
		if conn then
			table.insert(dummyHookConns, conn)
			track(conn)
		end
	end

	add(model.AncestryChanged:Connect(function(_, parent)
		if not parent then
			onDummyLost("dummy removed")
			return
		end
		local live = getLive()
		if live and not model:IsDescendantOf(live) then
			onDummyLost("dummy removed")
		end
	end))

	pcall(function()
		add(model.Destroying:Connect(function()
			onDummyLost("dummy destroyed")
		end))
	end)

	local function hookHumanoid(hum)
		if not hum then
			return
		end
		add(hum.Died:Connect(function()
			onDummyLost("dummy died")
		end))
		add(hum.HealthChanged:Connect(function(h)
			if h <= 0 then
				onDummyLost("dummy died")
			end
		end))
	end

	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum then
		hookHumanoid(hum)
	else
		add(model.ChildAdded:Connect(function(ch)
			if ch:IsA("Humanoid") then
				hookHumanoid(ch)
			end
		end))
	end
end

local function hookLive(live)
	disconnectList(liveHookConns)
	liveHookConns = {}
	if not live then
		return
	end

	local function add(conn)
		if conn then
			table.insert(liveHookConns, conn)
			track(conn)
		end
	end

	add(live.ChildAdded:Connect(function(child)
		if not isTestDummyName(child.Name) then
			return
		end
		task.wait()
		hookDummy(child)
		if enabled then
			approachAfterRespawn(child)
		end
	end))

	add(live.ChildRemoved:Connect(function(child)
		if isTestDummyName(child.Name) then
			onDummyLost("dummy removed")
		end
	end))

	local existing = getDummyModel()
	if existing and existing.Parent == live then
		hookDummy(existing)
	end
end

do
	local liveNow = getLive()
	if liveNow then
		hookLive(liveNow)
	end
	track(workspaceRef.ChildAdded:Connect(function(ch)
		if ch.Name == "Live" then
			hookLive(ch)
		end
	end))
end

local function guiParent()

		local ok, hui = pcall(gethui)
		if ok and hui then
			return hui
		end
	end

		local probe = Instance.new("Folder")
		probe.Name = "DF_Probe"
		probe.Parent = cg
		probe:Destroy()
		canCore = true
	end)
	if canCore then
		return cg
	end

	local pg = player:FindFirstChild("PlayerGui")
	if not pg then
		pg = player:WaitForChild("PlayerGui", 5)
	end
	return pg
end

local function destroyOld(parent)
	if not parent then
		return
	end
	for _, child in ipairs(parent:GetChildren()) do
		if child.Name == "DummyFollowGui" then
			pcall(function()
				child:Destroy()
			end)
		end
	end
end

	if typeof(gethui) == "function" then
		destroyOld(gethui())
	end
end)

	destroyOld(svc("CoreGui"))
end)

	local pg = player:FindFirstChild("PlayerGui")
	if pg then
		destroyOld(pg)
	end
end)

	pcall(function()
		TweenService:Create(obj, ti, props):Play()
	end)
end

		pill.Text = "ON"

		tween(pill, quick, { BackgroundColor3 = Color3.fromRGB(48, 52, 70) })
	end
end

local function hover(btn, rest, over)
	track(btn.MouseEnter:Connect(function()
		tween(btn, quick, { BackgroundColor3 = over })
	end))
	track(btn.MouseLeave:Connect(function()
		tween(btn, quick, { BackgroundColor3 = rest })
	end))
end

if not dragOk then
	local dragging = false
	local dragStart, startPos

	local function beginDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = panel.Position
		end
	end

	local function updateDrag(input)
		if not dragging then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			local d = input.Position - dragStart
			panel.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end

	local function endDrag(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end

	track(header.InputBegan:Connect(beginDrag))
	track(UserInputService.InputChanged:Connect(updateDrag))
	track(UserInputService.InputEnded:Connect(endDrag))
	pcall(function()
		track(UserInputService.TouchMoved:Connect(updateDrag))
		track(UserInputService.TouchEnded:Connect(endDrag))
	end)
end

local function paintStatus()
	if not gui.Parent then
		return
	end

	local dist = currentDistance()
	if dist then
		distLabel.Text = string.format("Distance  ·  %.1f studs", dist)
	elseif waitingRespawn then
		distLabel.Text = "Distance  ·  waiting for respawn"
	else
		distLabel.Text = "Distance  ·  dummy not found"
	end

	if automatic and enabled then
		local remain = math.max(0, nextAutoAt - os.clock())
		autoLabel.Text = string.format("Automatic  ·  next check in %ds", math.floor(remain + 0.5))
	elseif automatic and not enabled then
		autoLabel.Text = "Automatic  ·  waiting for Enable"
	else
		autoLabel.Text = "Automatic  ·  off"
	end

	if not enabled then
		statusLabel.Text = "Disabled"
		statusDot.BackgroundColor3 = Color3.fromRGB(110, 116, 140)
	elseif waitingRespawn or not getDummyLowerTorso() then
		statusLabel.Text = "Waiting for respawn"
		statusDot.BackgroundColor3 = Color3.fromRGB(230, 170, 70)
	elseif walking then
		statusLabel.Text = "Walking  ·  " .. lastWalkReason
		statusDot.BackgroundColor3 = Color3.fromRGB(88, 160, 255)
	else
		statusLabel.Text = "Watching"
		statusDot.BackgroundColor3 = Color3.fromRGB(90, 210, 140)
	end
end

bindClick(enableBtn, function()
	enabled = not enabled
	paintToggle(enablePill, enabled)
	if enabled then
		enableHint.Text = "Following, including after respawns"
		nextAutoAt = os.clock() + AUTO_INTERVAL
		local dummy = getDummyModel()
		if dummy and isDummyAlive(dummy) then
			local dist = currentDistance()
			if waitingRespawn or (dist and dist > FOLLOW_MIN) then
				approachAfterRespawn(dummy)
			end
		else
			waitingRespawn = true
			lastWalkReason = "waiting respawn"
		end
	else
		enableHint.Text = "Walk when dummy is 10-20 studs out"
		stopWalking()
		lastWalkReason = "idle"
	end
	paintStatus()
end)

	automatic = not automatic
	paintToggle(autoPill, automatic)
	if automatic then
		autoHint.Text = "Checking dummy drift every 2 minutes"

		local dummy = getDummyLowerTorso()
		if dummy then
			lastDummyPos = dummy.Position
		end
	else
		autoHint.Text = "Recheck drift every 2 minutes"
	end
	paintStatus()
end)

task.spawn(function()
	local trigger = pickFollowTrigger()
	while running and gui.Parent do
		if enabled and not walking then
			local ok, dist = pcall(currentDistance)
			if ok and dist and dist > trigger then
				walkUntilClose(string.format("enable  %.0f studs", trigger))
				trigger = pickFollowTrigger()
			elseif ok and dist and dist <= STOP_DISTANCE then
				trigger = pickFollowTrigger()
			end
		end
		task.wait(SCAN_RATE)
	end
end)

task.spawn(function()
	while running and gui.Parent do
		task.wait(0.25)
		if automatic and enabled and os.clock() >= nextAutoAt then
			nextAutoAt = os.clock() + AUTO_INTERVAL

			local dummy = getDummyLowerTorso()
			local torso = getPlayerTorso()
			if dummy and torso then
				local dist = planarDistance(torso.Position, dummy.Position)
				local moved = 0
				if lastDummyPos then
					moved = (dummy.Position - lastDummyPos).Magnitude
				end
				lastDummyPos = dummy.Position

				local driftNeed = pickAutoDrift()
				if dist >= driftNeed or moved >= driftNeed then
					walkUntilClose(string.format("auto  %.0f studs", driftNeed))
				end
			end
		end
	end
end)

-- Backup poll in case ChildAdded is missed after a respawn
task.spawn(function()
	while running and gui.Parent do
		local model = getDummyModel()
		local alive = model and isDummyAlive(model)
		if not alive then
			if not waitingRespawn then
				onDummyLost("dummy missing")
			end
		else
			if model ~= seenDummy then
				hookDummy(model)
				if enabled then
					approachAfterRespawn(model)
				end
			elseif waitingRespawn then
				if enabled then
					approachAfterRespawn(model)
				else
					waitingRespawn = false
				end
			end
		end
		task.wait(0.35)
	end
end)

	if not running then

	end
	pcall(stepWalk)

	if uiOpen and statusAcc >= 0.2 then
		statusAcc = 0
		pcall(paintStatus)
	end
end))

----------------------------------------------------------------------
-- Cleanup (re-execute in Delta replaces this instance)
----------------------------------------------------------------------
local function cleanup()
	running = false
	enabled = false
	automatic = false
	stopWalking()
	disconnectList(connections)
	disconnectList(dummyHookConns)
	disconnectList(liveHookConns)
	connections = {}
	dummyHookConns = {}
	liveHookConns = {}
	pcall(function()
		gui:Destroy()
	end)
	G.__DummyFollowKill = nil
end

G.__DummyFollowKill = cleanup

pcall(function()
	gui.Destroying:Connect(function()
		running = false
		stopWalking()
	end)
end)
