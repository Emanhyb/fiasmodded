--[[
	Dummy Follow  ·  Delta Executor
	How to run: open Delta → paste this whole script → Execute.

	Enable     — while ON, if Test_Dummy is more than 1-2 studs from
	             you, walk toward it until it is 1 stud from your torso.
	Automatic  — while ON (and Enable is ON), every 2 minutes check
	             whether the dummy has drifted ~1-2 studs away.
	             If it has, walk in until you are 1 stud from your torso.
	X          — hides the panel. Tap the chip to reopen it.
	Aim Lock   — face the dummy after he respawns (or when you re-enter
	             range). Does NOT lock the camera. Fight / Tool 1 turns
	             on shiftlock; this just points you at him first so
	             shiftlock does not snap you the wrong way.
	Auto Click — standalone toggle. While ON, repeatedly left-clicks
	             the center of the screen and keeps the cursor locked
	             there. Speed slider sets clicks per second (1–20 CPS).
	             If the dummy is more than 1-2 studs out: unequip
	             Backpack.Fight (Tool 1) and pause clicking. When
	             the dummy is in range: re-equip Tool 1 and resume clicking.

	Respawn   — First spawn is workspace.Live.Test_Dummy. After it
	             respawns (private server) the path is workspace.Test_Dummy.
	             Both are watched. If Enable is ON, walk back to 1 stud.

	Feet on head — if your foot touches Test_Dummy.Head (too close /
	             standing on it), unequip Tool 1, hold S to back up,
	             then walk back into 1-2 stud attack range.

	Hell Gloves — Boxing Gloves SKIN (not the weapon). Buy Boxing
	             Gloves, then put Hell / Blessed / etc. on them.
	             Using Hell Gloves needs 20,000 leaderstats.Kills.
	             Panel shows kills / 20000. Stay on Jeet Kune Do
	             until then — bare Boxing Gloves is only 4 damage.
]]

local function isTestDummyName(name)
	if not name then
		return false
	end
	local n = string.lower(name)
	return n == "test_dummy" or n == "testdummy"
end

local function isDummyAlive(model)
	if not model or not model.Parent then
		return false
	end
	if not model:IsDescendantOf(workspaceRef) then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum and hum.Health <= 0 then
		return false
	end
	return true
end

local function considerDummy(inst, list, seen)
	if not inst or seen[inst] then
		return
	end
	if not isTestDummyName(inst.Name) then
		return
	end
	seen[inst] = true
	list[#list + 1] = inst
end

-- First spawn: workspace.Live.Test_Dummy
-- After first private-server respawn: workspace.Test_Dummy
local function getDummyModel()
	local list = {}
	local seen = {}
	considerDummy(workspaceRef:FindFirstChild("Test_Dummy"), list, seen)
	local live = getLive()
	if live then
		considerDummy(live:FindFirstChild("Test_Dummy"), list, seen)
		for _, child in ipairs(live:GetChildren()) do
			considerDummy(child, list, seen)
		end
	end
	for _, child in ipairs(workspaceRef:GetChildren()) do
		considerDummy(child, list, seen)
	end
	for i = 1, #list do
		if isDummyAlive(list[i]) then
			return list[i]
		end
	end
	if #list > 0 then
		return list[1]
	end
	return nil
end

local function getDummyHead()

	if not dummy or not isDummyAlive(dummy) then
		return nil

		or dummy:FindFirstChild("head")

local FOOT_NAMES = {
	LeftFoot = true,
	RightFoot = true,
	LeftLowerLeg = true,
	RightLowerLeg = true,
	["Left Leg"] = true,
	["Right Leg"] = true,
}

-- True if a foot is on the dummy head, or you spawned stacked on top of it.
local function feetTouchDummyHead()
	local head = getDummyHead()
	local char = getCharacter()
	if not head or not char then
		return false
	end
	for _, child in ipairs(char:GetDescendants()) do
		if child:IsA("BasePart") and FOOT_NAMES[child.Name] then
			local d = (child.Position - head.Position).Magnitude
			local reach = 2.1
			pcall(function()
				reach = math.max(1.8, child.Size.Magnitude * 0.45 + head.Size.Y * 0.6)
			end)
			if d <= reach then
				return true
			end
		end
	end
	local root = getRoot()
	if root then
		local dx = root.Position.X - head.Position.X
		local dz = root.Position.Z - head.Position.Z
		local xz = math.sqrt(dx * dx + dz * dz)
		local dy = root.Position.Y - head.Position.Y
		if xz <= 3.2 and dy > 0.4 and dy < 9 then
			return true
		end
	end
	return false
end

local function planarDistance(a, b)
	local dx = a.X - b.X
	local dz = a.Z - b.Z
	return math.sqrt(dx * dx + dz * dz)
end

local function currentDistance()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()
	if not torso or not dummy then
		return nil
	end
	return planarDistance(torso.Position, dummy.Position)
end

local function isFightTool(inst)
	if not inst or not inst:IsA("Tool") then
		return false
	end
	local n = string.lower(tostring(inst.Name))
	return n == "fight" or n == "1" or n == "tool 1" or n == "tool1"
end

-- Fight is hotbar Tool 1. Prefer that name, else the first Tool in Backpack.
local function getFightTool()
	local char = getCharacter()
	if char then
		for _, child in ipairs(char:GetChildren()) do
			if isFightTool(child) then
				return child
			end
		end
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") then
				return child
			end
		end
	end
	local bag = getBackpack()
	if bag then
		local named = bag:FindFirstChild("Fight")
		if named and named:IsA("Tool") then
			return named
		end
		for _, child in ipairs(bag:GetChildren()) do
			if child:IsA("Tool") then
				return child
			end
		end
	end
	return nil
end

local function equippedTool()
	local char = getCharacter()
	if not char then
		return nil
	end
	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Tool") then
			return child
		end
	end
	return nil
end

local function fightIsEquipped()
	local char = getCharacter()
	if not char then
		return false
	end
	for _, child in ipairs(char:GetChildren()) do
		if isFightTool(child) then
			return true
		end
	end
	return false
end

-- ONLY called when dummy is more than 1-2 studs out. Never while in range.
local function unequipFight()
	local hum = getHumanoid()
	if hum then
		pcall(function()
			hum:UnequipTools()
		end)
	end
	local char = getCharacter()
	local bag = getBackpack()
	if char and bag then
		for _, child in ipairs(char:GetChildren()) do
			if child:IsA("Tool") then
				pcall(function()
					child.Parent = bag
				end)
			end
		end
	end
end

-- Press hotbar 1 once. Never spam — 1 toggles Tool 1 off if already held.
local function pressKey1Once()
	pcall(function()
		if VirtualInputManager then
			VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.One, false, game)
			VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.One, false, game)
		end
	end)
	if typeof(keypress) == "function" then
		pcall(keypress, 0x31)
		if typeof(keyrelease) == "function" then
			pcall(keyrelease, 0x31)
		end
	end
end

local lastEquipTry = 0
local useKey1Next = false

-- Only while IN RANGE and Fight is NOT equipped. Never runs out of range.
local function ensureFightEquipped()
	if fightIsEquipped() then
		useKey1Next = false
		return
	end
	local now = os.clock()
	if now - lastEquipTry < 0.45 then
		return
	end
	lastEquipTry = now

	if useKey1Next then
		pressKey1Once()
		useKey1Next = false
		return
	end

	local hum = getHumanoid()
	local tool = getFightTool()
	if hum and tool and tool.Parent ~= getCharacter() then
		pcall(function()
			hum:EquipTool(tool)
		end)
	end
	useKey1Next = true
end

-- Turn character + camera toward the dummy ONCE so Fight shiftlock
-- inherits that heading. Never hold the camera every frame.
local function faceDummy()
	local root = getRoot()
	local dummy = getDummyLowerTorso()
	if not root or not dummy then
		return false
	end
	local lookAt = Vector3.new(dummy.Position.X, root.Position.Y, dummy.Position.Z)
	if (lookAt - root.Position).Magnitude < 0.08 then
		return false
	end
	local ok = pcall(function()
		root.CFrame = CFrame.lookAt(root.Position, lookAt)
	end)
	if not ok then
		return false
	end
	local cam = workspaceRef.CurrentCamera
	if cam then
		pcall(function()
			local look = root.CFrame.LookVector
			cam.CFrame = CFrame.new(cam.CFrame.Position, cam.CFrame.Position + look)
		end)
	end
	aimEngaged = true
	lastFaceAt = os.clock()
	return true
end

local function requestFace()
	if not aimLockOn then
		return
	end
	needFace = true
	faceUntil = os.clock() + 1.25
end

local function stepFaceDummy()
	if not running or not aimLockOn then
		needFace = false
		if not aimLockOn then
			aimEngaged = false
		end
		return
	end
	if not needFace then
		return
	end
	if os.clock() > faceUntil then
		needFace = false
		return
	end
	if os.clock() - lastFaceAt < 0.12 then
		return
	end
	if faceDummy() then
		needFace = false
	end
end

-- true = in range, false = clearly out, nil = unknown (do not unequip)
local function dummyRangeState()
	local dist = currentDistance()
	if dist == nil then
		return nil
	end
	if dist <= FOLLOW_MAX then
		return true
	end
	return false
end

local function updateFightAndClick()
	if not running then
		clickAllowed = false
		return
	end
	if backingUp then
		clickAllowed = false
		return
	end
	if not (enabled or clickerOn) then
		clickAllowed = false
		return
	end

	local inRange = dummyRangeState()
	if inRange == true then
		-- Stay in range: keep checking. Equip only if Tool 1 is not out.
		ensureFightEquipped()
		fightState = "in"
		clickAllowed = clickerOn
	elseif inRange == false then
		useKey1Next = false
		if fightIsEquipped() or equippedTool() then
			unequipFight()
		end
		fightState = "out"
		clickAllowed = false
	else
		if fightState == "out" then
			clickAllowed = false
		else
			clickAllowed = clickerOn
		end
	end
end

local function pickFollowTrigger()
	return FOLLOW_MIN + math.random() * (FOLLOW_MAX - FOLLOW_MIN)
end

local function pickAutoDrift()
	return AUTO_DRIFT_MIN + math.random() * (AUTO_DRIFT_MAX - AUTO_DRIFT_MIN)
end

----------------------------------------------------------------------
-- Movement  (Humanoid:Move every frame — MoveTo is often cancelled)
----------------------------------------------------------------------
local function haltHumanoid()
	local humanoid = getHumanoid()
	if humanoid then
		pcall(function()
			humanoid:Move(Vector3.zero)
		end)
	end
end

local function stopWalking()
	walking = false
	haltHumanoid()
end

local function walkUntilClose(reason)
	walking = true
	lastWalkReason = reason
	lastMoveToAt = 0
end

-- Stand 1.7 studs in front of the dummy spawn so the next respawn
-- is already in attack range (not under your feet).
local function parkForNextKill(anchorPos)
	if not enabled then
		return
	end
	local root = getRoot()
	if not root or not anchorPos then
		return
	end
	local away = Vector3.new(root.Position.X - anchorPos.X, 0, root.Position.Z - anchorPos.Z)
	if away.Magnitude < 0.25 then
		local look = root.CFrame.LookVector
		away = Vector3.new(-look.X, 0, -look.Z)
	end
	if away.Magnitude < 0.25 then
		away = Vector3.new(0, 0, 1)
	end
	local dest = Vector3.new(
		anchorPos.X + away.Unit.X * 1.7,
		root.Position.Y,
		anchorPos.Z + away.Unit.Z * 1.7
	)
	local lookAt = Vector3.new(anchorPos.X, root.Position.Y, anchorPos.Z)
	pcall(function()
		root.CFrame = CFrame.lookAt(dest, lookAt)
	end)
end

local function setKeyS(down)
	pcall(function()
		if VirtualInputManager then
			VirtualInputManager:SendKeyEvent(down, Enum.KeyCode.S, false, game)
		end
	end)
	if down then
		if typeof(keypress) == "function" then
			pcall(keypress, 0x53)
		end
	else
		if typeof(keyrelease) == "function" then
			pcall(keyrelease, 0x53)
		end
	end
end

-- Dummy respawned under you: unequip Tool 1, hold S to step off the head,
-- then walk back into 1-2 stud attack range so punches land again.
local function startBackup()
	if backingUp then
		return
	end
	backingUp = true
	clickAllowed = false
	walking = false
	useKey1Next = false
	fightState = "out"
	unequipFight()
	lastWalkReason = "off head"

	task.spawn(function()
		local root = getRoot()
		local head = getDummyHead()
		local anchor = lastDummyPos
		if head then
			anchor = head.Position
		end
		if root and anchor then
			parkForNextKill(anchor)
		end
		if feetTouchDummyHead() then
			setKeyS(true)
			local t0 = os.clock()
			while running and os.clock() - t0 < 0.35 do
				root = getRoot()
				local hum = getHumanoid()
				if hum and root then
					pcall(function()
						hum:Move(-root.CFrame.LookVector)
					end)
				end
				if not feetTouchDummyHead() then
					break
				end
				task.wait(0.04)
			end
			setKeyS(false)
			haltHumanoid()
		end
		backingUp = false
		requestFace()
		faceDummy()
		if running and enabled and not feetTouchDummyHead() then
			local dist = currentDistance()
			if dist and dist > FOLLOW_MAX then
				walkUntilClose("reposition")
			end
		end
	end)
end

local function checkStandOnHead()
	if backingUp then
		return
	end
	if not (enabled or clickerOn) then
		return
	end
	if feetTouchDummyHead() then
		startBackup()
	end
end

local function stepWalk()
	if backingUp then
		return
	end
	if not walking or not enabled then
		return
	end

	local humanoid = getHumanoid()
	local root = getRoot()
	local torso = getPlayerTorso()
	local dummy = getDummyLowerTorso()

	if not humanoid or humanoid.Health <= 0 or not root or not torso then
		walking = false
		haltHumanoid()
		return
	end

	if not dummy then
		waitingRespawn = true
		lastWalkReason = "waiting respawn"
		walking = false
		haltHumanoid()
		return
	end

	local dist = planarDistance(torso.Position, dummy.Position)
	if dist <= STOP_DISTANCE then
		haltHumanoid()
		walking = false
		lastWalkReason = "arrived"
		requestFace()
		return
	end

	local flat = Vector3.new(
		dummy.Position.X - root.Position.X,
		0,
		dummy.Position.Z - root.Position.Z
	)
	local mag = flat.Magnitude
	if mag > 0.05 then
		pcall(function()
			humanoid:Move(flat.Unit)
		end)
		local now = os.clock()
		if now - lastMoveToAt > 0.25 then
			lastMoveToAt = now
			pcall(function()
				humanoid:MoveTo(dummy.Position)
			end)
		end
	end
end

local function alignCameraToRoot()
	local root = getRoot()
	local cam = workspaceRef.CurrentCamera
	if not root or not cam then
		return
	end
	pcall(function()
		local look = root.CFrame.LookVector
		cam.CFrame = CFrame.new(cam.CFrame.Position, cam.CFrame.Position + look)
	end)
end

----------------------------------------------------------------------
-- Dummy respawn watcher
----------------------------------------------------------------------
local function onDummyLost(reason)
	local deathPos = lastDummyPos
	if seenDummy then
		local part = seenDummy:FindFirstChild("LowerTorso")
			or seenDummy:FindFirstChild("Head")
			or seenDummy:FindFirstChild("HumanoidRootPart")
		if part then
			deathPos = part.Position
		end
	end
	if deathPos then
		lastDummyPos = deathPos
	end
	seenDummy = nil
	waitingRespawn = true
	if walking then
		walking = false
		haltHumanoid()
	end
	aimEngaged = false
	clickAllowed = false
	useKey1Next = false
	fightState = "out"
	unequipFight()
	lastWalkReason = reason or "waiting respawn"
	-- Step to the 1.7-stud pocket so the next dummy is already in range.
	if deathPos then
		parkForNextKill(deathPos)
		if aimLockOn then
			alignCameraToRoot()
		end
	end
end

local function approachAfterRespawn(model)
	approachToken = approachToken + 1
	local myToken = approachToken
	task.spawn(function()
		local deadline = os.clock() + RESPAWN_WAIT
		local part = nil
		while running and myToken == approachToken and os.clock() < deadline do
			if not model or not model.Parent then
				return
			end
			part = model:FindFirstChild("LowerTorso")
			if part then
				break
			end
			task.wait(0.1)
		end
		if myToken ~= approachToken or not running then
			return
		end
		if not part then
			part = dummyAimPart(model)
		end
		if not part then
			return
		end
		task.wait(RESPAWN_SETTLE)
		if myToken ~= approachToken or not running then
			return
		end
		if not model.Parent or not isDummyAlive(model) then
			return
		end
		waitingRespawn = false
		seenDummy = model
		lastDummyPos = part.Position
		-- Face him before Fight shiftlock turns back on.
		requestFace()
		faceDummy()
		if not enabled then
			return
		end
		if feetTouchDummyHead() then
			startBackup()
		else
			local dist = currentDistance()
			if dist and dist > FOLLOW_MAX then
				walkUntilClose("respawn")
			end
			-- already in 1-2 stud pocket: stay put and start hitting
		end
	end)
end

local function hookDummy(model)
	disconnectList(dummyHookConns)
	dummyHookConns = {}
	if not model then
		return
	end
	seenDummy = model
	waitingRespawn = false

	local function add(conn)
		if conn then
			table.insert(dummyHookConns, conn)
			track(conn)
		end
	end

	add(model.AncestryChanged:Connect(function(_, parent)
		if not parent then
			onDummyLost("dummy removed")
			return
		end
		if not model:IsDescendantOf(workspaceRef) then
			onDummyLost("dummy removed")
		end
	end))

	pcall(function()
		add(model.Destroying:Connect(function()
			onDummyLost("dummy destroyed")
		end))
	end)

	local function hookHumanoid(hum)
		if not hum then
			return
		end
		add(hum.Died:Connect(function()
			onDummyLost("dummy died")
		end))
		add(hum.HealthChanged:Connect(function(h)
			if h <= 0 then
				onDummyLost("dummy died")
			end
		end))
	end

	local hum = model:FindFirstChildOfClass("Humanoid")
	if hum then
		hookHumanoid(hum)
	else
		add(model.ChildAdded:Connect(function(ch)
			if ch:IsA("Humanoid") then
				hookHumanoid(ch)
			end
		end))
	end

	local function hookHead(head)
		if not head or not head:IsA("BasePart") then
			return
		end
		add(head.Touched:Connect(function(hit)
			if hit and FOOT_NAMES[hit.Name] then
				local char = getCharacter()
				if char and hit:IsDescendantOf(char) then
					startBackup()
				end
			end
		end))
	end

	local head = model:FindFirstChild("Head") or model:FindFirstChild("head")
	if head then
		hookHead(head)
	else
		add(model.ChildAdded:Connect(function(ch)
			if ch.Name == "Head" or ch.Name == "head" then
				hookHead(ch)
			end
		end))
	end

	if feetTouchDummyHead() and (enabled or clickerOn) then
		startBackup()
	end
end

local function hookLive(live)
	disconnectList(liveHookConns)
	liveHookConns = {}
	if not live then
		return
	end

	local function add(conn)
		if conn then
			table.insert(liveHookConns, conn)
			track(conn)
		end
	end

	add(live.ChildAdded:Connect(function(child)
		if not isTestDummyName(child.Name) then
			return
		end
		task.wait()
		hookDummy(child)
		if enabled then
			approachAfterRespawn(child)
		end
	end))

	add(live.ChildRemoved:Connect(function(child)
		if isTestDummyName(child.Name) then
			task.wait()
			if not getDummyModel() then
				onDummyLost("dummy removed")
			end
		end
	end))

	local existing = getDummyModel()
	if existing and existing.Parent == live then
		hookDummy(existing)
	end
end

local function setClickSpeedFromAlpha(alpha)
	if alpha < 0 then
		alpha = 0
	elseif alpha > 1 then
		alpha = 1
	end
	clickCps = math.floor(CLICK_CPS_MIN + alpha * (CLICK_CPS_MAX - CLICK_CPS_MIN) + 0.5)
	if clickCps < CLICK_CPS_MIN then
		clickCps = CLICK_CPS_MIN
	elseif clickCps > CLICK_CPS_MAX then
		clickCps = CLICK_CPS_MAX
	end
	speedValue.Text = tostring(clickCps) .. " CPS"
	speedFill.Size = UDim2.new(alpha, 0, 1, 0)
	speedKnob.Position = UDim2.new(alpha, 0, 0.5, 0)
	if clickerOn then
		clickHint.Text = "Clicking  ·  " .. tostring(clickCps) .. " CPS"
	end
end

		local absX = speedTrack.AbsolutePosition.X
		local w = speedTrack.AbsoluteSize.X
		if w < 1 then
			return
		end
		setClickSpeedFromAlpha((x - absX) / w)
	end

	local function beginSlide(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			if panelDrag then
				pcall(function()
					panelDrag.Enabled = false
				end)
			end
			setFromX(input.Position.X)
		end
	end

	local function moveSlide(input)
		if not sliding then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			setFromX(input.Position.X)
		end
	end

	local function endSlide(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
			if panelDrag then
				pcall(function()
					panelDrag.Enabled = true
				end)
			end
		end
	end

	track(speedTrack.InputBegan:Connect(beginSlide))
	track(speedKnob.InputBegan:Connect(beginSlide))
	track(UserInputService.InputChanged:Connect(moveSlide))
	track(UserInputService.InputEnded:Connect(endSlide))

			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end

	track(header.InputBegan:Connect(beginDrag))
	track(UserInputService.InputChanged:Connect(updateDrag))
	track(UserInputService.InputEnded:Connect(endDrag))
	pcall(function()
		track(UserInputService.TouchMoved:Connect(updateDrag))
		track(UserInputService.TouchEnded:Connect(endDrag))
	end)
end

local function paintStatus()
	if not gui.Parent then
		return
	end

	local dummyModel = getDummyModel()
	subtitle.Text = dummyPathLabel(dummyModel)

	local dist = currentDistance()
	if dist then
		distLabel.Text = string.format("Distance  ·  %.1f studs", dist)
	elseif waitingRespawn then
		distLabel.Text = "Distance  ·  waiting for respawn"
	else
		distLabel.Text = "Distance  ·  dummy not found"
	end

	if automatic and enabled then
		local remain = math.max(0, nextAutoAt - os.clock())
		autoLabel.Text = string.format("Automatic  ·  next check in %ds", math.floor(remain + 0.5))
	elseif automatic and not enabled then
		autoLabel.Text = "Automatic  ·  waiting for Enable"
	elseif clickerOn and clickAllowed then
		autoLabel.Text = string.format("Clicker  ·  center  %d CPS", clickCps)
	elseif clickerOn then
		autoLabel.Text = "Clicker  ·  waiting for range"
	else
		autoLabel.Text = "Automatic  ·  off"
	end

	local kills = getKills()
	if kills then
		if kills >= HELL_SKIN_KILLS then
			killLabel.Text = string.format("Hell Gloves  ·  %d / %d  READY", kills, HELL_SKIN_KILLS)
			killLabel.TextColor3 = Color3.fromRGB(110, 220, 150)
		else
			killLabel.Text = string.format(
				"Hell Gloves  ·  %d / %d  (%d left)",
				kills,
				HELL_SKIN_KILLS,
				HELL_SKIN_KILLS - kills
			)
			killLabel.TextColor3 = Color3.fromRGB(230, 170, 70)
		end
	else
		killLabel.Text = "Hell Gloves  ·  kills -- / 20000"
		killLabel.TextColor3 = Color3.fromRGB(150, 156, 180)
	end

	if backingUp then
		statusLabel.Text = "Off head  ·  holding S"
		statusDot.BackgroundColor3 = Color3.fromRGB(255, 200, 80)
	elseif aimLockOn and aimEngaged then
		statusLabel.Text = "Faced dummy  ·  shiftlock"
		statusDot.BackgroundColor3 = Color3.fromRGB(255, 120, 130)
	elseif clickerOn and clickAllowed then
		statusLabel.Text = "Fight  ·  clicking"
		statusDot.BackgroundColor3 = Color3.fromRGB(255, 180, 80)
	elseif clickerOn then
		statusLabel.Text = "Fight unequipped  ·  out of range"
		statusDot.BackgroundColor3 = Color3.fromRGB(230, 170, 70)
	elseif not enabled then
		statusLabel.Text = "Disabled"
		statusDot.BackgroundColor3 = Color3.fromRGB(110, 116, 140)
	elseif waitingRespawn or not getDummyLowerTorso() then
		statusLabel.Text = "Waiting for respawn"
		statusDot.BackgroundColor3 = Color3.fromRGB(230, 170, 70)
	elseif walking then
		statusLabel.Text = "Walking  ·  " .. lastWalkReason
		statusDot.BackgroundColor3 = Color3.fromRGB(88, 160, 255)
	else
		statusLabel.Text = "Watching"
		statusDot.BackgroundColor3 = Color3.fromRGB(90, 210, 140)
	end
end

----------------------------------------------------------------------
-- Auto clicker (standalone — does not need Enable)
----------------------------------------------------------------------
local function pointOver(guiObj, point)
	if not guiObj or not guiObj.Visible then
		return false
	end
	local p = guiObj.AbsolutePosition
	local s = guiObj.AbsoluteSize
	return point.X >= p.X and point.X <= p.X + s.X
		and point.Y >= p.Y and point.Y <= p.Y + s.Y
end

local function screenCenter()
	local cam = workspaceRef.CurrentCamera
	if cam then
		return cam.ViewportSize * 0.5
	end
	return Vector2.new(400, 300)
end

local function lockCursorCenter()
	local pos = screenCenter()
	if typeof(mousemoveabs) == "function" then
		pcall(mousemoveabs, pos.X, pos.Y)
	end
	return pos
end

local function centerOverOurGui()
	local loc = screenCenter()
	local inset = Vector2.new(0, 0)
	pcall(function()
		inset = GuiService:GetGuiInset()
	end)
	local adj = Vector2.new(loc.X - inset.X, loc.Y - inset.Y)
	if uiOpen and panel.Visible and (pointOver(panel, loc) or pointOver(panel, adj)) then
		return true
	end
	if restore.Visible and (pointOver(restore, loc) or pointOver(restore, adj)) then
		return true
	end
	return false
end

local function fireClick()
	if not clickerOn or not running or not clickAllowed then
		return
	end
	local pos = lockCursorCenter()
	if centerOverOurGui() then
		return
	end

	if VirtualInputManager then
		pcall(function()
			VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 1)
			VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 1)
		end)
		return
	end
	if typeof(mouse1click) == "function" then
		pcall(mouse1click)
		return
	end
	if typeof(mouse1press) == "function" then
		pcall(mouse1press)
		task.wait(0.015)
		if typeof(mouse1release) == "function" then
			pcall(mouse1release)
		end
	end
end

-- Loops

task.spawn(function()
	local trigger = pickFollowTrigger()
	while running and gui.Parent do
		if enabled and not walking then
			local ok, dist = pcall(currentDistance)
			if ok and dist and dist > trigger then
				walkUntilClose(string.format("enable  %.0f studs", trigger))
				trigger = pickFollowTrigger()
			elseif ok and dist and dist <= STOP_DISTANCE then
				trigger = pickFollowTrigger()
			end
		end
		task.wait(SCAN_RATE)
	end
end)

task.spawn(function()
	while running and gui.Parent do
		task.wait(0.25)
		if automatic and enabled and os.clock() >= nextAutoAt then
			nextAutoAt = os.clock() + AUTO_INTERVAL

			local dummy = getDummyLowerTorso()
			local torso = getPlayerTorso()
			if dummy and torso then
				local dist = planarDistance(torso.Position, dummy.Position)
				local moved = 0
				if lastDummyPos then
					moved = (dummy.Position - lastDummyPos).Magnitude
				end
				lastDummyPos = dummy.Position

				local driftNeed = pickAutoDrift()
				if dist >= driftNeed or moved >= driftNeed then
					walkUntilClose(string.format("auto  %.0f studs", driftNeed))
				end
			end
		end
	end
end)

-- Backup poll in case ChildAdded is missed after a respawn
task.spawn(function()
	while running and gui.Parent do
		local model = getDummyModel()
		local alive = model and isDummyAlive(model)
		if not alive then
			if not waitingRespawn then
				onDummyLost("dummy missing")
			end
		else
			if model ~= seenDummy then
				hookDummy(model)
				if enabled then
					approachAfterRespawn(model)
				end
			elseif waitingRespawn then
				if enabled then
					approachAfterRespawn(model)
				else
					waitingRespawn = false
				end
			end
		end
		task.wait(0.35)
	end
end)

task.spawn(function()
	while running and gui.Parent do
		if clickerOn then
			pcall(fireClick)
			task.wait(clickInterval())
		else
			task.wait(0.12)
		end
	end
end)

-- Cleanup (re-execute in Delta replaces this instance)
----------------------------------------------------------------------
local function cleanup()
	running = false
	enabled = false
	automatic = false
	aimLockOn = false
	aimEngaged = false
	clickerOn = false
	clickAllowed = false
	backingUp = false
	pcall(setKeyS, false)
	pcall(unequipFight)
	stopWalking()
	pcall(function()
		RunService:UnbindFromRenderStep(AIM_STEP_NAME)
	end)
	disconnectList(connections)
	disconnectList(dummyHookConns)
	disconnectList(liveHookConns)
	disconnectList(workspaceDummyConns)
	connections = {}
	dummyHookConns = {}
	liveHookConns = {}
	workspaceDummyConns = {}
	pcall(function()
		gui:Destroy()
	end)
	G.__DummyFollowKill = nil
end

		running = false
		stopWalking()
	end)
end)
